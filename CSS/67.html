Keyframe Animation Syntax 

.box {
    animation: move 3s ease-in-out infinite;
}

@keyframes move {

    0% {
        transform: translateX(0);
    }

    50% {
        transform: translateX(100px);
    }

    100% {
        transform: translateX(0);
    }

}
