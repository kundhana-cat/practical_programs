Transformation Syntax 

/* 2D Transformations */

.box {
    transform: translate(50px, 20px);
    transform: rotate(180deg);
    transform: scale(1.5);
    transform: skew(20deg);
}


/* 3D Transformations */

.box {
    transform: rotateX(180deg);
    transform: rotateY(180deg);
    transform: rotateZ(180deg);
    transform: scale3d(1.5, 1.5, 1.5);
}
